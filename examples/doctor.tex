\section{理论基础}

\begin{lemma}\label{lem:rank}
设 $A$ 为 $m \times n$ 矩阵，$B$ 为 $n \times p$ 矩阵，则 $\operatorname{rank}(AB) \leq \min\{\operatorname{rank}(A), \operatorname{rank}(B)\}$。
\end{lemma}

\begin{proof}
由矩阵乘法的定义可知 $AB$ 的列空间包含于 $A$ 的列空间，$AB$ 的行空间包含于 $B$ 的行空间，结论显然。
\end{proof}

\begin{corollary}
若 $A$ 为可逆矩阵，则 $\operatorname{rank}(AB) = \operatorname{rank}(B)$。
\end{corollary}

\begin{proof}
由\cref{lem:rank}知 $\operatorname{rank}(AB) \leq \operatorname{rank}(B)$，又 $B = A^{-1}(AB)$，故 $\operatorname{rank}(B) \leq \operatorname{rank}(AB)$，从而等号成立。
\end{proof}

\section{算法环境示例}

\begin{algorithm}[!ht]
\caption{快速排序算法}
\KwIn{数组 $A[1..n]$}
\KwOut{排序后的数组 $A$}
\If{$n \leq 1$}{
    \Return $A$\;
}
$p \leftarrow A[1]$\;
$i \leftarrow 1$, $j \leftarrow n$\;
\While{$i < j$}{
    \While{$i < j$ \textbf{and} $A[j] \geq p$}{
        $j \leftarrow j - 1$\;
    }
    $A[i] \leftarrow A[j]$\;
    \While{$i < j$ \textbf{and} $A[i] \leq p$}{
        $i \leftarrow i + 1$\;
    }
    $A[j] \leftarrow A[i]$\;
}
$A[i] \leftarrow p$\;
递归排序 $A[1..i-1]$ 和 $A[i+1..n]$\;
\Return $A$\;
\end{algorithm}
