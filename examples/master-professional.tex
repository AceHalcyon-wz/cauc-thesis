\section{算法实现}

\begin{algorithm}[!ht]
\caption{梯度下降优化算法}
\KwIn{目标函数 $f(\theta)$，学习率 $\eta$，迭代次数 $T$}
\KwOut{优化后的参数 $\theta^*$}
初始化 $\theta_0$\;
\For{$t = 1, 2, \ldots, T$}{
    计算梯度 $g_t \leftarrow \nabla f(\theta_{t-1})$\;
    更新参数 $\theta_t \leftarrow \theta_{t-1} - \eta g_t$\;
    \If{收敛条件满足}{
        \textbf{break}\;
    }
}
\Return $\theta_T$\;
\end{algorithm}
