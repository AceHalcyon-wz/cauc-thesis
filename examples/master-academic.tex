\section{定理环境示例}

\begin{theorem}
设 $A$ 为 $n$ 阶实对称矩阵，则 $A$ 的特征值均为实数。
\end{theorem}

\begin{proof}
设 $\lambda$ 为 $A$ 的特征值，$x$ 为对应的特征向量，则 $Ax = \lambda x$。
取共轭转置得 $x^* A^* = \bar{\lambda} x^*$，由于 $A$ 为实对称矩阵，$A^* = A$，
故 $x^* A = \bar{\lambda} x^*$。左乘 $x$ 得 $x^* A x = \bar{\lambda} x^* x$，
又 $x^* A x = \lambda x^* x$，故 $(\lambda - \bar{\lambda}) x^* x = 0$，
因 $x \neq 0$，故 $\lambda = \bar{\lambda}$，即 $\lambda$ 为实数。
\end{proof}

\section{算法环境示例}

\begin{algorithm}[!ht]
\caption{二分查找算法}
\KwIn{有序数组 $A[1..n]$，目标值 $target$}
\KwOut{目标值的索引，若不存在则返回 $-1$}
$left \leftarrow 1$, $right \leftarrow n$\;
\While{$left \leq right$}{
    $mid \leftarrow \lfloor (left + right) / 2 \rfloor$\;
    \eIf{$A[mid] = target$}{
        \Return $mid$\;
    }{
        \eIf{$A[mid] < target$}{
            $left \leftarrow mid + 1$\;
        }{
            $right \leftarrow mid - 1$\;
        }
    }
}
\Return $-1$\;
\end{algorithm}
