% 章节起始页页眉控制：headeronplain=true 时显示页眉（与原始模板一致），false 时不显示。
% 条件在样式使用时求值，保证导言区或 cauc_thesis.cfg 中的 \caucsetup{headeronplain=false} 生效。
\newcommand{\cauc@setplainstyle}{%
  \fancypagestyle{plain}{%
    \fancyhf{}%
    \fancyfoot[C]{\thepage}%
    \ifcauc@headeronplain
      \fancyhead[C]{\cauc@header@text}%
      \renewcommand{\headrulewidth}{0.5pt}%
    \else
      \renewcommand{\headrulewidth}{0pt}%
    \fi
  }%
}
\cauc@setplainstyle

% 双面打印时页眉与页码在奇偶页均按版心居中排列（CE/CO）
\ifcauc@twoside
  \fancypagestyle{blank}{%
    \fancyhf{}%
    \ifcauc@blankpagestandard
      \fancyhead[CE,CO]{\cauc@header@text}%
      \fancyfoot[CE,CO]{\thepage}%
      \renewcommand{\headrulewidth}{0.5pt}%
    \else
      \renewcommand{\headrulewidth}{0pt}%
    \fi
  }
  \renewcommand{\cleardoublepage}{%
    \clearpage
    \if@twoside
      \ifodd\c@page\else
        \hbox{}%
        \thispagestyle{blank}%
        \newpage
      \fi
    \fi
  }
  \pagestyle{fancy}
  \fancyhf{}
  \fancyhead[CE,CO]{\cauc@header@text}
  \fancyfoot[CE,CO]{\thepage}
  \renewcommand{\headrulewidth}{0.5pt}
\fi

% 图表标题格式在 \AtBeginDocument 中应用，使导言区
% \caucsetup{captionlabelsep=...} 也能生效。
\newcommand{\cauc@setcaptions}{%
  % 使用星号形式，避免文档未使用某类浮动体时 caption 宏包报告 "Unused \captionsetup"。
  \ifx\cauc@captionlabelsep\@empty
    \captionsetup*[figure]{font={normalsize,rm},name={图},labelsep=quad}
    \captionsetup*[subfigure]{font={normalsize,rm},name={图},labelsep=quad}
    \captionsetup*[table]{labelsep=space}
  \else
    \captionsetup*[figure]{font={normalsize,rm},name={图},labelsep=\cauc@captionlabelsep}
    \captionsetup*[subfigure]{font={normalsize,rm},name={图},labelsep=\cauc@captionlabelsep}
    \captionsetup*[table]{labelsep=\cauc@captionlabelsep}
  \fi
}
